%% ---------------------------------------------------------------------------
%% Custom title page.
%%
%% ltx-talk's stock \maketitle fills the whole frame and centres vertically, so
%% the old decks' habit of adding a "Material adapted from ..." block after
%% \maketitle collides with it. We therefore define our own title frame, which
%% also lets us make the (otherwise bare) title slide our own.
%%
%%   \coursetitlepage{<main title>}{<subtitle>}{<attribution>}
%% ---------------------------------------------------------------------------
\newcommand{\coursetitlepage}[3]{%
  \begin{frame}
    \centering
    \vfill
    {\color{bostonuniversityred}\Huge\bfseries #1}\\[0.5em]
    {\Large #2}\\[2.5em]
    {\large Felipe Meneguzzi \qquad Yaji Sripada}\\[0.4em]
    {\small\ttfamily felipe.meneguzzi@abdn.ac.uk \quad yaji.sripada@abdn.ac.uk}\\[0.4em]
    University of Aberdeen\\[2.5em]
    {\small #3}%
    \vfill
  \end{frame}%
}

%% Section dividers.
%%
%% The old decks opened each section with an \AtBeginSection outline frame
%% (\tableofcontents[currentsection]). In ltx-talk 0.5.0 \tableofcontents
%% corrupts the PDF tag tree (mismatched para/text-unit hooks) and cannot be
%% produced as tagged output, so we cannot reproduce that outline as-is.
%%
%% Instead, \heading{Title} starts the section AND emits a clean, tagging-safe
%% divider frame showing the section title. Replace \section{Title} with
%% \heading{Title} in the decks. (Plain \section{...} still works if you want a
%% silent section with no divider.)
\newcommand{\heading}[1]{%
  \section{#1}%
  \begin{frame}%
    \vfill
    \begin{center}
      {\color{bostonuniversityred}\Huge\bfseries #1}
    \end{center}
    \vfill
  \end{frame}%
}
